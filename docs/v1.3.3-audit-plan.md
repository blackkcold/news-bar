## 一、风险分析与应对策略

### P0 修复的风险

| 修复项 | 潜在风险 | 应对策略 |
|---|---|---|
| `[self]` → `[weak self]` | self 为 nil 时 API Key 保存静默失败 | 添加 `else { NSLog }` 日志 + 重试逻辑 |

| AboutTab 缓存清除 | 创建新实例无法清除主 orchestrator 状态 | 注入主 orchestrator 引用（见 Phase 0.4 修正） |

### P1 修复的风险

| 修复项 | 潜在风险 | 应对策略 |
|---|---|---|
| NewsOrchestrator 重构 | 状态机回归 bug | 先写测试再重构 |
| HTTPClient 提取 | 不同服务的超时/header 需求不同 | 使用配置结构体 |

### Phase 0: 紧急安全修复（1 天）

#### 0.1 修复代理 SHA256 验证漏洞

**文件**: `Services/UpdateChecker.swift`

**问题**: 当前 `fetchSHA256`（:234）从 `sha256Asset.browser_download_url` 获取 SHA256 文件，该 URL 同样经过代理。如果代理被篡改，可同时提供恶意 DMG 和匹配的 SHA256，签名验证形同虚设。

**方案 A（推荐）— 从 GitHub API 获取 SHA256 URL**:

```swift
private func fetchSHA256(from release: GitHubRelease, viaProxy: Bool) async -> String? {
    guard let sha256Asset = release.assets.first(where: { $0.name.hasSuffix(".sha256") }) else {
        return nil
    }
    
    // 如果通过代理下载 DMG，则 SHA256 从 GitHub 官方获取
    let sha256URLString: String
    if viaProxy {
        // 从 browser_download_url 中提取 GitHub 原始路径
        // 代理 URL 格式: https://gh-proxy.com/https://github.com/...
        // 提取 https://github.com/... 部分
        sha256URLString = sha256Asset.browser_download_url
            .replacingOccurrences(of: "https://gh-proxy.com/", with: "")
            .replacingOccurrences(of: "https://gh.api.99988866.xyz/", with: "")
    } else {
        sha256URLString = sha256Asset.browser_download_url
    }
    
    guard let sha256URL = URL(string: sha256URLString) else { return nil }
    // ... 其余逻辑不变
}
```

**方案 B — 从 GitHub API 独立获取 release 信息**:

```swift
// 在 downloadUpdate 中，先从 GitHub API 获取 release（含 SHA256 URL）
// 再从代理下载 DMG
// 最后用 GitHub API 的 SHA256 验证代理下载的 DMG
```

**回退方案**: 保留代理但默认关闭，设置中添加"使用镜像源"开关。使用代理时 SHA256 必须从 GitHub 官方获取。

#### 0.2 SSRF 防护 — 警告模式

**文件**: `Services/SecurityPolicies.swift`

**修正**: 当前 `validateRSSURL` 返回 `Bool`，需要新增返回类型：

```swift
enum RSSURLValidation {
    case valid
    case blocked(reason: String)
    case warning(reason: String)
}

static func validateRSSURL(_ urlString: String) -> RSSURLValidation {
    guard let url = validateURL(urlString), let host = url.host else {
        return .blocked(reason: "无效的 URL")
    }
    let blockedDomains: Set<String> = ["localhost", "127.0.0.1", "0.0.0.0", "::1"]
    if blockedDomains.contains(host.lowercased()) {
        return .blocked(reason: "不允许访问本地地址")
    }
    if isPrivateIP(host) {
        return .warning(reason: "该地址可能是内网地址，请确认来源可信")
    }
    return .valid
}

private static func isPrivateIP(_ host: String) -> Bool {
    // 10.0.0.0/8
    if let ip = IPv4Address(host), ip.isPrivate { return true }
    // 备选：正则检查
    let patterns = [
        "^10\\.",                           // 10.0.0.0/8
        "^172\\.(1[6-9]|2[0-9]|3[01])\\.",  // 172.16.0.0/12
        "^192\\.168\\.",                     // 192.168.0.0/16
        "^169\\.254\\.",                     // 169.254.0.0/16 (link-local)
        "^fc00:",                            // fc00::/7
        "^fe80:",                            // fe80::/10
    ]
    return patterns.contains { host.range(of: $0, options: .regularExpression) != nil }
}
```

**调用方更新**: RSSTab.swift 中根据返回值显示警告对话框（`.warning`）或阻止（`.blocked`）。

#### 2.1 消除 NewsOrchestrator 刷新重复

**方案**: 提取公共刷新核心方法

```swift
private func doRefresh(
    settings: AppSettings,
    trigger: RefreshLog.Trigger,
    isManual: Bool
) async {
    guard !isRefreshing else { return }
    
    let aiBefore = logStateLabel(aiSummaryState)
    if isManual {
        manualRefreshWarning = await rateLimiter.manualRefreshWarning()
    }
    
    settings.recordRefresh()
    isRefreshing = true
    markSources(settings.activeSources, as: .loading)
    defer { isRefreshing = false }
    
    let previousSummaryState = aiSummaryState
    let apiKey = settings.cachedAPIKey ?? ""
    if settings.aiSummaryEnabled {
        aiSummaryState = apiKey.isEmpty ? .noKey : .fetching
    }
    
    // 并行获取所有源
    await withTaskGroup(of: Void.self) { group in
        group.addTask { await self.fetchWeibo() }
        group.addTask { await self.fetchBilibili() }
        for source in settings.activeSources where !source.isBuiltIn {
            group.addTask { await self.fetchRSS(source: source) }
        }
    }
    
    if isManual {
        await rateLimiter.recordManualRefresh()
        manualRefreshWarning = await rateLimiter.manualRefreshWarning()
    }
    
    await handleAISummary(settings: settings, apiKey: apiKey, previousState: previousSummaryState)
    await recordRefreshLog(settings: settings, trigger: trigger, aiBefore: aiBefore)
}

func refreshIfNeeded(settings: AppSettings, trigger: RefreshLog.Trigger = .startup) async {
    await doRefresh(settings: settings, trigger: trigger, isManual: false)
}

func manualRefresh(settings: AppSettings) async {
    await doRefresh(settings: settings, trigger: .manual, isManual: true)
}
```

**额外收益**: 使用 `withTaskGroup` 并行获取，总刷新时间从累加变为最慢单源时间。

```swift
enum HTTPClient {
    struct Config {
        let timeout: TimeInterval
        let userAgent: String
        let extraHeaders: [String: String]
        
        static let weibo = Config(timeout: 10, userAgent: mobileUA, extraHeaders: [
            "Referer": "https://weibo.com/",
            "X-Requested-With": "XMLHttpRequest",
            "Cache-Control": "no-cache"
        ])
        static let bilibili = Config(timeout: 8, userAgent: desktopUA, extraHeaders: [
            "Referer": "https://www.bilibili.com"
        ])
        static let rss = Config(timeout: 8, userAgent: "NewsBar/1.0 (macOS; RSS Reader)", extraHeaders: [:])
        static let ai = Config(timeout: 30, userAgent: "NewsBar/1.0", extraHeaders: [:])
    }
    
    static func data(for url: URL, config: Config) async throws -> (Data, HTTPURLResponse) {
        var request = URLRequest(url: url)
        request.setValue(config.userAgent, forHTTPHeaderField: "User-Agent")
        for (key, value) in config.extraHeaders {
            request.setValue(value, forHTTPHeaderField: key)
        }
        request.timeoutInterval = config.timeout
        
        let (data, response) = try await URLSession.shared.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse else {
            throw NewsBarError.requestFailed
        }
        guard (200...299).contains(httpResponse.statusCode) else {
            throw NewsBarError.requestFailed
        }
        return (data, httpResponse)
    }
}
```
